\addscenariosection{1}{Clash/Alliance Scenario}{Mystical Foreland}{\images/pathfinding.png}

\begin{multicols}{2}

\textbf{Author:} BoberL

\textit{Colonization of the unseen lands went well, until your scouts brought news of unknown horse trails by the nearby lake.
You are not the only one looking for the Mystical Foreland.
Double the nightguard and move out at dawn.}  % no-check-caps

\subsection*{\MakeUppercase{Scenario Length}}
16 Rounds

\subsection*{\MakeUppercase{Player Setup}}
\textbf{Player Count:} 4

\textbf{Starting Resources:} 14~\svg{gold}, 3~\svg{building_materials}, 1~\svg{valuables}

\textbf{Starting Income:} 15~\svg{gold}, 0~\svg{building_materials}, 0~\svg{valuables}

\textbf{Starting Units:}

\begin{itemize}
  \item Search~(2) the \svg{bronze} Neutral Deck
  \item A ``Few'' chosen \svg{bronze} Units
\end{itemize}

\textbf{Town Buildings:} \svg{bronze} Dwelling

\subsection*{\MakeUppercase{Map Setup}}
Take the following Map Tiles and arrange them as shown in the Scenario map layout:

\begin{itemize}
  \item 4 × Starting (I) Map Tiles
  \item 8 × Far (II--III) Map Tiles
  \item 14 × Near (IV--V) Map Tiles (6 Land, 6 Sea, 2 Subterranean). These Tiles should not contain any Obelisks.
  \item 3 × Center (VI--VII) Map Tiles (1 Grail, 1 Sea, 1 Subterranean)
\end{itemize}

Green-tinted Tiles in the Scenario map layout contain \textbf{Creature Banks}.

\subsection*{\MakeUppercase{Victory Conditions}}
\begin{itemize}
  \item Obtain the Grail Token and bring it to your Faction Town.
  \item Defeat all remaining Players.
\end{itemize}

\subsection*{\MakeUppercase{Defeat Conditions}}
If no player brings the Grail Token to their Town, all players lose the Scenario.

\subsection*{\MakeUppercase{Timed Events}}
\textbf{\nth{4} Round:}
\begin{itemize}
  \item Each player may immediately \svg{reinforce} one \svg{bronze} Unit of their choice for free, or one \svg{silver} Unit of their choice at a cost reduced by 4~\svg{gold}.
\end{itemize}
\textbf{\nth{6}, \nth{10}, \nth{14} Rounds:}
\begin{itemize}
  \item Remove all Black Cubes from the map except those on Learning Stones.
\end{itemize}
\textbf{\nth{8} Round:}
\begin{itemize}
  \item Increase your \svg{gold}, \svg{building_materials} or \svg{valuables} income by 1 Cube.
\end{itemize}
\textbf{\nth{9} Round:}
\begin{itemize}
  \item Draw and resolve an additional Event Card. Drawer rotates clockwise after each, as normal.
\end{itemize}
\textbf{\nth{10} Round:}
\begin{itemize}
  \item From now on, the \textbf{Custom Combat Rule} applies (see Additional Rules below).
\end{itemize}
\textbf{\nth{12} Round:}
\begin{itemize}
  \item Each player may \svg{pay} 5~\svg{gold} to Search~(2) the \svg{artifact}, \svg{skill}, or \svg{spellpower} Deck.
    A player may do this up to 3 times, paying 5~\svg{gold} each time.
\end{itemize}

\subsection*{\MakeUppercase{Additional Rules}}

\subsubsection*{Setup Changes}

\begin{itemize}
  \item Before the game, remove the \textbf{Diplomacy} and \textbf{Pathfinding} Abilities, the \textbf{Den of Thieves}, \textbf{Mercenary Camp} and \textbf{Prison} Events, the \textbf{Ambassador's Sash} and \textbf{Diplomat's Ring} Artifacts.
  \item Level VII Neutral Combats cannot be skipped.
  \item Stepping on a Sanctuary Tile lets you Search(2) your Deck.
  \item Defeating an enemy Hero grants the winning Player 1~\svg{experience} and allows to Search(2) the Ability or Artifact Deck.
  \item Depending on the difficulty, players can choose their starting bonus:
  \begin{itemize}
    \item \textbf{Easy}: Roll 2~\svg{resource} or 2~\svg{treasure} and gain both bonuses \textbf{OR} Search(4) the \svg{artifact} Deck.
      \item \textbf{Normal}: Roll 2~\svg{resource} or 2~\svg{treasure} and gain one chosen bonus \textbf{OR} Draw 3 \svg{artifact} Cards and take one into your Hand. If any Relic is drawn this way, discard it and draw another.
      \item \textbf{Hard}: Roll 2~\svg{resource} Dice and gain one chosen bonus \textbf{OR} Search(2) the \svg{artifact} Deck. If any Relic \svg{artifact} is drawn this way, discard it and draw another.
      \item \textbf{Impossible}: Roll 1~\svg{resource} Die and gain its bonus \textbf{OR} Search(2) the \svg{artifact} Deck. If any Relic or Major \svg{artifact} is drawn this way, discard it and draw another.
    \end{itemize}
\end{itemize}

\subsubsection*{Map Constraints}

\begin{itemize}
  \item A Hero may not cross a \textbf{Red Border}, drawn as a red line between two Tiles in the Scenario map layout, by any means.
  \item The Grail Tile and the two Sea Tiles adjacent to it are also Subterranean, as marked in the Scenario map layout.
  \item Entering or exiting a Sea Tile only takes up an extra 1 \svg{movement}.
  \item If playing Alliance mode, alternate starting Tile placement such that allies do not start adjacent to each other.
  \item If playing Alliance mode, the team that's being attacked may \svg{pay} 10~\svg{gold} to move either Player to the defending town and allow them to use their Deck.
\end{itemize}

\subsubsection*{Grail Rules}

\begin{itemize}
  \item To conquer the Grail (VII), players fight the \textbf{Grail Army} instead of the regular Neutral army.
    It contains the regular Neutral Units plus an extra 1~\svg{azure} Unit and 1~\svg{golden} Unit.
    Enemy Units get a permanent \textbf{Stack Token} according to the difficulty:
    \begin{itemize}
      \item \textbf{Easy}: 1~\textbf{Stack Token} placed on any \svg{golden} Unit.
      \item \textbf{Normal}: 1~\textbf{Stack Token} placed on any \svg{azure} Unit.
      \item \textbf{Hard}: 2~\textbf{Stack Tokens} placed on 1~\svg{golden} Unit and 1~\svg{azure} Unit.
      \item \textbf{Impossible}: 2~\textbf{Stack Tokens} placed on any \svg{azure} Units.
    \end{itemize}
  \item A player who is defeated in Combat or surrenders drops the Grail Token on the Field where the Hero was defeated or surrendered.
  \item A player holding the Grail can move by an extra Tile.
  \item A player holding the Grail Token has their \svg{hand} Limit increased by 2.
  \item If a Hero holding the Grail Token uses Angel Wings to fly over impassable Tiles, Dimension Door, Fly or Town Portal, the Grail Token is placed on the Field the Hero occupied before the move.
\end{itemize}

\subsubsection*{Neutral Unit Recruitment}

\begin{itemize}
  \item Only if you meet the requirements below, once at any point during your Turn, you may \svg{pay} one of the following to Search~(2) the Neutral Units Deck:
    \begin{itemize}
      \item \svg{bronze} \textbf{Dwelling}: 3~\svg{gold}, 1~\svg{building_materials} for \svg{bronze}
      \item \svg{silver} \textbf{Dwelling}: 4~\svg{gold}, 2~\svg{building_materials} for \svg{silver}
      \item \svg{golden} \textbf{Dwelling}: 5~\svg{gold}, 3~\svg{building_materials}, 1~\svg{valuables} for \svg{golden}
      \item \svg{golden} \textbf{Dwelling}, \textbf{Citadel}: 8~\svg{gold}, 4~\svg{building_materials}, 2~\svg{valuables} for \svg{azure}
    \end{itemize}
    You may then choose one of the Units \svg{pay} and pay its full cost to recruit it.
\end{itemize}

\subsubsection*{Field Rewards}

\begin{itemize}
  \item \textbf{Temple of the Sea}: 15~\svg{gold}.
    Search~(3) the \svg{artifact} Deck \textbf{twice}.
    You may remove 2 Statistic Cards from your Deck or Discard Pile and take their Empowered counterparts into your hand in their stead.
  \item \textbf{Level VII Town:} You may \svg{recruit} and \svg{reinforce} one of the Units from the defending Town for half its cost, excluding \svg{valuables} (rounded down). \textit{Example: After defeating a Castle Army, you may now \svg{recruit} and \svg{reinforce} Archangels for 25~\svg{gold} and 3~\svg{valuables}.}
  \item \textbf{Cyclops Stockpile}: 10~\svg{building_materials}, 5~\svg{valuables}. Your Units gain a permanent +1~\svg{attack} and +1~\svg{initiative}, including Units you Recruit later.
\end{itemize}

\subsubsection*{Custom Combat Rule}

Starting in the \nth{10} Round, the following rules apply:

\begin{itemize}
  \item Place one extra \textbf{Stack Token} on each enemy Neutral \svg{golden} Unit. Do not place it in Combat against the \textbf{Grail Army} or another player.
  \item For each extra \textbf{Stack Token}, the Hero gains additional +1~\svg{movement}.
  \item For every \svg{golden} Unit you defeat in a Neutral Combat, gain 5~\svg{gold} and either 2~\svg{building_materials} or 1~\svg{valuables}.
\end{itemize}
\end{multicols}
