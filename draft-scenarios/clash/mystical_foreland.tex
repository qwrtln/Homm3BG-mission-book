\addscenariosection{1}{Clash/Alliance Scenario}{Mystical Foreland}{\images/logistics.png}

\begin{multicols*}{2}

\textbf{Author:} BoberL

\textit{Colonization of the unseen lands has been going well, until your scouts have brought news about unknown horse trails next to the nearby lake. You come to a conclusion, that you're not the only person looking for the Mystical Foreland. Your next move is to double the nightguard and get moving at dawn.}  % no-check-caps

\subsection*{\MakeUppercase{Required Expansions:}}

\begin{itemize}
  \item{\textcolor{conflux}{\textbf{Conflux Expansion}}}
  \item{\textcolor{cove}{\textbf{Cove Expansion}} (\& \textcolor{navalbattles}{\textbf{Naval Battles Expansion}} for tile variety})
  \item{\textcolor{stronghold}{\textbf{Stronghold Expansion}} (\& \textcolor{stretchgoals2}{\textbf{Stretch Goals Regular}} for tile variety})
  \item{\textcolor{inferno}{\textbf{Inferno Expansion}}}
  \item{\textcolor{rampart}{\textbf{Rampart Expansion}}}
\end{itemize}

\subsection*{\MakeUppercase{Scenario Length}}
14 Rounds

\subsection*{\MakeUppercase{Player Setup}}
\textbf{Player Count:} 4

\textbf{Starting Resources:} 13 \svg{gold}, 3 \svg{building_materials}, 1 \svg{valuables}

\textbf{Starting Income:} 10 \svg{gold}, 0 \svg{building_materials}, 0 \svg{valuables}

\textbf{Starting Units:}

\begin{itemize}
  \item Search~(2) the \svg{bronze} Neutral Deck
  \item 1 × A Few \svg{bronze} Units of your choice
\end{itemize}

\textbf{Town Buildings:} \svg{bronze} Dwelling

\subsection*{\MakeUppercase{Map Setup}}
Take the following Map Tiles and arrange them as shown in the Scenario map layout:

\begin{itemize}
  \item 4 × Starting (I) Map Tiles
  \item 8 × Far (II--III) Map Tiles
  \item 10 × Near (IV--V) Map Tiles (2 Normal, 6 Sea, 2 Underground). These tiles should not contain any Obelisks.
  \item 3 × Center (VI--VII) Map Tiles (1 Grail, 1 Sea, 1 Underground)
\end{itemize}

\subsection*{\MakeUppercase{Victory Conditions}}
The game ends when a player brings the grail back to their town.

\subsection*{\MakeUppercase{Timed Events}}
\textbf{\nth{4} Round:}
\begin{itemize}
  \item Each player may instantly \textbf{Reinforce} a single \svg{bronze} Unit of their choice for free or a \svg{silver} Unit for 4 \svg{gold} less.
\end{itemize}
\textbf{\nth{6} Round:}
\begin{itemize}
  \item Remove all Black Cubes from the map except those on Learning Stones.
\end{itemize}
\textbf{\nth{9} Round:}
\begin{itemize}
  \item Resolve 2 Event Cards.
\end{itemize}
\textbf{\nth{10} Round:}
\begin{itemize}
  \item From now on, every neutral \svg{golden} Units gain an extra \textbf{Upgrade token}. Ignore this rule, when fighting the \textbf{Grail Army}.
\end{itemize}
\textbf{\nth{12} Round:}
\begin{itemize}
  \item Each player can \svg{pay} 5 \svg{gold} to Search(2) the \svg{artifact} Deck, \svg{skill} or \svg{spellpower} up to 3 times.
\end{itemize}

\subsection*{\MakeUppercase{Additional Rules}}
\begin{itemize}
  \item Players cannot enter or exit Sea tiles from \textcolor{red}{\textbf{Red tiles}}.
  \item \textcolor{blue}{\textbf{Blue tiles}} are the Underground tiles.
  \item \textcolor{rampart}{\textbf{Lime tile}} must contain the Grail.
  \item Tiles that have the \textcolor{inferno}{\textbf{Red cubes}} contain \textbf{Creature Banks}.
  \item Instead of fighting the regular enemies to conquer the Grail (T7), players instead fight the \textbf{Grail Army}. Besides the regular enemies, it also contains an extra 1 \svg{azure} Unit and 1 \svg{golden} Unit. Each enemy unit gets a permanent \textbf{Stack token}.
  \item \textbf{Sea Temple}: 15 \svg{gold}. Search(3) the \svg{artifact} Deck \textbf{twice} You may replace 2 Stat cards from your Deck or Discard for their upgraded versions. Take both these cards into your hand.	
  \item \textbf{Level VII Town:} You may Recruit and Reinforce one of the Units from the defending town for half its \svg{gold} (Rounded down). 
 \item \textbf{Cyclops Stockpile}: 10 \svg{building_materials} 5 \svg{valuables}. All of your units now have permanent +1 \svg{attack} and +1 \svg{initiative}.
  \item Level VII fights cannot be skipped.
  \item Players that are defeated in combat or surrender drop their Grail at the tile they're currently at.
  \item Players holding the Grail have their \svg{hand} increased by 2.
  \item You may Search(2) the Neutral deck at \textbf{any point during your turn} for according prices:


\begin{itemize} 
  \item 3 \svg{gold} 1 \svg{building_materials} for \svg{bronze},
  \item 5 \svg{gold} 2 \svg{building_materials} 1 \svg{valuables} for \svg{silver},
  \item 9 \svg{gold} 4 \svg{building_materials} 2 \svg{valuables} for \svg{golden},
  \item 15 \svg{gold} 6 \svg{building_materials} 4 \svg{valuables} for \svg{azure}.
\end{itemize}

  \item You can \svg{pay} to recruit these Units for their full cost.
\end{itemize}

\columnbreak

\begin{itemize}
  \item Remove Diplomancy ability cards, Events that let you recruit Neutral Units and Diplomancy related \svg{artifact} from the game.
  \item If a Hero with the Grail Token uses Dimension Door, Fly, or Town Portal, the Grail Token is dropped on the origin Field before the Hero moves.
  \item \svg{golden} Neutral Units fought during and after \textbf{Round 10} gain an extra \textbf{Stack token}. Ignore this rule, when fighting the \textbf{Grail Army}. 
  \item If you're playing a 2v2 scenario, the opposing teams should have a player each on each \textcolor{red}{outlined} and not outlined Starting (I) Tiles.
\end{itemize}
